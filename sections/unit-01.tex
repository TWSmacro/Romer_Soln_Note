\begin{problem}{1}[\textbf{Basic properties of growth rates.}]
Use the fact that the growth rate of a variable equals the time derivative of its log to show:
\begin{enumerate}[label=(\alph*)]
    \item The growth rate of the product of two variables equals the sum of their growth rates.
    That is, if \(Z\left(t\right)=X\left(t\right)Y\left(t\right)\), then
    \begin{align*}
        \dfrac{\dot Z\left(t\right)}{Z\left(t\right)} = 
        \dfrac{\dot X\left(t\right)}{X\left(t\right)} + 
        \dfrac{\dot Y\left(t\right)}{Y\left(t\right)}
        \problemeqtag
    \end{align*}
    \item The growth rate of the ratio of two variables equals the difference of their growth rates.
    That is, if \(Z\left(t\right)=X\left(t\right)/Y\left(t\right)\), then
    \begin{align*}
        \dfrac{\dot Z\left(t\right)}{Z\left(t\right)} = 
        \dfrac{\dot X\left(t\right)}{X\left(t\right)} - 
        \dfrac{\dot Y\left(t\right)}{Y\left(t\right)}
        \problemeqtag
    \end{align*}
    \item If \(Z\left(t\right)=a\,X\left(t\right)^{b}\), then
    \begin{align*}
        \dfrac{\dot Z\left(t\right)}{Z\left(t\right)} =
        b\cdot\dfrac{\dot X\left(t\right)}{X\left(t\right)}
        \problemeqtag
    \end{align*}
\end{enumerate}
\end{problem}

\begin{solution}{1.1.a}
For any positive differentiable variable \(X\left(t\right)\), define its instantaneous growth rate by
\begin{align*}
    g_X\left(t\right)\equiv \frac{\dot X\left(t\right)}{X\left(t\right)}
    =\frac{\mathrm d \ln X\left(t\right)}{\d t}.
    \problemeqtag\label{eq:p11-growth-definition}
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item If \(Z\left(t\right)=X\left(t\right)Y\left(t\right)\), then
    \begin{align*}
        \ln Z\left(t\right)&=\ln X\left(t\right)+\ln Y\left(t\right),\\
        \frac{\d\ln Z\left(t\right)}{\d t}
        &=\frac{\d\ln X\left(t\right)}{\d t}
        +\frac{\d\ln Y\left(t\right)}{\d t}
    \end{align*}
    Therefore,
    \begin{align*}
        \boxed{
        \frac{\dot Z\left(t\right)}{Z\left(t\right)}
        =\frac{\dot X\left(t\right)}{X\left(t\right)}
        +\frac{\dot Y\left(t\right)}{Y\left(t\right)}}
    \end{align*}

    \item If \(Z\left(t\right)=X\left(t\right)/Y\left(t\right)\), then
    \begin{align*}
        \ln Z\left(t\right)&=\ln X\left(t\right)-\ln Y\left(t\right),\\
        \frac{\d\ln Z\left(t\right)}{\d t}
        &=\frac{\d\ln X\left(t\right)}{\d t}
        -\frac{\d\ln Y\left(t\right)}{\d t}
    \end{align*}
    Hence,
    \begin{align*}
        \boxed{
        \frac{\dot Z\left(t\right)}{Z\left(t\right)}
        =\frac{\dot X\left(t\right)}{X\left(t\right)}
        -\frac{\dot Y\left(t\right)}{Y\left(t\right)}}
    \end{align*}

    \item Assume \(a>0\) and let \(Z\left(t\right)=aX\left(t\right)^\alpha\), where \(a\) and
    \(\alpha\) are constants. Taking logs and then differentiating gives
    \begin{align*}
        \ln Z\left(t\right)&=\ln a+\alpha\ln X\left(t\right),\\
        \frac{\d\ln Z\left(t\right)}{\d t}
        &=\frac{\d}{\d t}\bigl[\ln a+\alpha\ln X\left(t\right)\bigr]\\
        &=0+\alpha\frac{\d\ln X\left(t\right)}{\d t}
    \end{align*}
    Using the growth-rate identity in \eqref{eq:p11-growth-definition} for
    both \(Z\left(t\right)\) and \(X\left(t\right)\), we obtain
    \begin{align*}
        \boxed{
        \frac{\dot Z\left(t\right)}{Z\left(t\right)}
        =\alpha\frac{\dot X\left(t\right)}{X\left(t\right)}}
    \end{align*}
    Thus a time-invariant multiplicative constant changes the level of a variable,
    but not its growth rate.
\end{enumerate}
\end{solution}

\begin{note}[Why logarithms work]
    Because instantaneous growth in continuous time is naturally expressed in exponential
    form, \(\lim_{m\to\infty}(1+g/m)^{mt}=e^{gt}\), the rules above reflect a useful
    duality: logarithms turn products into sums, while exponentials turn sums into products.
\end{note}

\begin{note}[Four growth-rate rules]
    For products, add the growth rates. For ratios, subtract them. For powers,
    multiply the growth rate by the exponent. Multiplying a variable by a constant
    does not change its growth rate.
\end{note}

\begin{note}[Sums and differences]
    Sums and differences also have growth-rate formulas, but there is no analogous
    unweighted rule. For example, the growth rate of \(X+Y\) is a level-weighted
    average of the growth rates of \(X\) and \(Y\).
\end{note}

\begin{problem}{2}
Suppose that the growth rate of some variable, \(X\), is constant and equal to \(a>0\) from time 0
to time \(t_1\); drops to 0 at time \(t_1\); rises gradually from 0 to \(a\) from time \(t_1\)
to time \(t_2\); and is constant and equal to \(a\) after time \(t_2\).
\begin{enumerate}[label=(\alph*)]
    \item Sketch a graph of the growth rate of \(X\) as a function of time.
    \item Sketch a graph of \(\ln X\) as a function of time.
\end{enumerate}
\end{problem}

\begin{solution}
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{4pt}
Let
\begin{align*}
    \gamma_X\left(t\right)\equiv\frac{\dot X\left(t\right)}{X\left(t\right)}
    =\frac{\d\ln X\left(t\right)}{\d t}.
    \problemeqtag\label{eq:p12-growth-definition}
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item The question does not specify the exact shape of the gradual increase between
    \(t_1\) and \(t_2\). For a concrete sketch, suppose it is linear. Then
    \begin{align*}
        \gamma_X\left(t\right)=
        \begin{cases}
            a, & 0\leq t<t_1,\\[3pt]
            \dfrac{a(t-t_1)}{t_2-t_1}, & t_1\leq t\leq t_2,\\[7pt]
            a, & t>t_2.
        \end{cases}
        \problemeqtag\label{eq:p12-growth-path}
    \end{align*}
    Thus the growth rate is initially horizontal at \(a\), jumps down to zero at
    \(t_1\), rises to \(a\), and is horizontal again after \(t_2\).

    \item Since the slope of \(\ln X\left(t\right)\) equals \(\gamma_X\left(t\right)\),
    \begin{align*}
        \ln X\left(t\right)-\ln X(0)=\int_0^t\gamma_X(\tau)\,\d\tau.
        \problemeqtag\label{eq:p12-log-integral}
    \end{align*}
    Substituting the linear path in \eqref{eq:p12-growth-path} into
    \eqref{eq:p12-log-integral} gives
    \begin{align*}
        \ln X\left(t\right)=
        \begin{cases}
            \ln X(0)+at,
                &0\leq t\leq t_1,\\[3pt]
            \ln X(0)+at_1+
                \dfrac{a(t-t_1)^2}{2(t_2-t_1)},
                &t_1\leq t\leq t_2,\\[9pt]
            \ln X(0)+at_1+\dfrac{a(t_2-t_1)}{2}+a(t-t_2),
                &t\geq t_2.
        \end{cases}
        \problemeqtag
    \end{align*}
    Hence \(\ln X\left(t\right)\) is linear with slope \(a\) before \(t_1\). It remains
    continuous at \(t_1\), but its slope drops to zero. Between \(t_1\) and \(t_2\)
    it is increasing and convex, and after \(t_2\) it is again linear with slope \(a\).
\end{enumerate}

\begin{note}[Reading a log-level graph]
    A jump in a growth rate changes the slope of the log-level path, not its level.
    Thus \(\ln X\left(t\right)\) has a kink at \(t_1\), not a jump or an inflection point.
\end{note}

\begin{problem}{3}
Describe how, if at all, each of the following developments affects the break-even and actual
investment lines in our basic diagram for the Solow model:
\begin{enumerate}[label=(\alph*)]
    \item The rate of depreciation falls.
    \item The rate of technological progress rises.
    \item The production function is Cobb--Douglas, \(f\left(k\right)=k^{\alpha}\), and
    capital's share, \(\alpha\), rises.
    \item Workers exert more effort, so that output per unit of effective labor for a given value
    of capital per unit of effective labor is higher than before.
\end{enumerate}
\end{problem}

\begin{solution}
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{4pt}
The law of motion for capital per unit of effective labor is
\begin{align*}
    \dot k=sf(k)-(n+g+\delta)k.
    \problemeqtag\label{eq:p13-law-of-motion}
\end{align*}
The balanced-growth-path value \(k^*\) is determined by
\begin{align*}
    sf(k^*)=(n+g+\delta)k^*.
    \problemeqtag\label{eq:p13-steady-state}
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item By \eqref{eq:p13-law-of-motion}, a fall in \(\delta\) reduces
    \(n+g+\delta\), so the break-even investment
    line becomes flatter. The actual investment curve \(sf(k)\) is unchanged. At the old
    \(k^*\), actual investment now exceeds break-even investment, and therefore
    \begin{align*}
        \boxed{k^*_{\mathrm{new}}>k^*_{\mathrm{old}}}
    \end{align*}

    \item A rise in \(g\) increases \(n+g+\delta\), so the break-even investment line
    becomes steeper. The actual investment curve is unchanged. At the old \(k^*\), break-even
    investment now exceeds actual investment, and therefore
    \begin{align*}
        \boxed{k^*_{\mathrm{new}}<k^*_{\mathrm{old}}}
    \end{align*}

    \item With \(f(k)=k^\alpha\), actual investment is \(sk^\alpha\), while the
    break-even line is unaffected. For a fixed \(k\),
    \begin{align*}
        \frac{\partial(sk^\alpha)}{\partial\alpha}=sk^\alpha\ln k.
        \problemeqtag
    \end{align*}
    Thus an increase in \(\alpha\) lowers actual investment when \(0<k<1\), leaves it
    unchanged at \(k=1\), and raises it when \(k>1\). The actual investment curve
    therefore rotates around \(k=1\); it does not shift uniformly upward or downward.

    Solving the steady-state condition \eqref{eq:p13-steady-state} for the
    Cobb--Douglas case gives
    \begin{align*}
        k^*=\left(\frac{s}{n+g+\delta}\right)^{1/(1-\alpha)}.
        \problemeqtag
    \end{align*}
    Taking logs and differentiating with respect to \(\alpha\) gives
    \begin{align*}
        \frac{\partial\ln k^*}{\partial\alpha}
        =\frac{\ln\!\left(s/(n+g+\delta)\right)}{(1-\alpha)^2}.
        \problemeqtag
    \end{align*}
    Consequently,
    \begin{align*}
        \boxed{
        \begin{array}{lll}
            s>n+g+\delta &\Longrightarrow& k^*\text{ rises},\\
            s<n+g+\delta &\Longrightarrow& k^*\text{ falls},\\
            s=n+g+\delta &\Longrightarrow& k^*=1\text{ and is unchanged}.
        \end{array}}
    \end{align*}
    Without additional information, the effect on \(k^*\) is ambiguous.

    \item Let the higher effort level raise output at every relevant \(k\), so that the
    new production function satisfies \(\widetilde f(k)>f(k)\). Actual investment shifts
    upward from \(sf(k)\) to \(s\widetilde f(k)\), while the break-even line is unchanged.
    Hence
    \begin{align*}
        \boxed{k^*_{\mathrm{new}}>k^*_{\mathrm{old}}}
    \end{align*}
\end{enumerate}

    First identify which curve changes. Parameters \(n\), \(g\), and \(\delta\) rotate

    In part (c), a higher \(\alpha\) rotates rather than uniformly shifts \(sk^\alpha\):

    when \(k>1\).
\end{note}

\begin{problem}{4}
Consider an economy with technological progress but without population growth that is on its balanced
growth path. Now suppose there is a one-time jump in the number of workers.
\begin{enumerate}[label=(\alph*)]
    \item At the time of the jump, does output per unit of effective labor rise, fall, or stay
    the same? Why?
    \item After the initial change (if any) in output per unit of effective labor when the new
    workers appear, is there any further change in output per unit of effective labor? If so,
    does it rise or fall? Why?
    \item Once the economy has again reached a balanced growth path, is output per unit of
    effective labor higher, lower, or the same as it was before the new workers appeared? Why?
\end{enumerate}
\end{problem}

\begin{solution}
Let the one-time increase in labor occur at \(t_0\). Immediately before the shock,
the economy is on a balanced growth path, and
\begin{align*}
    k(t_0^-)=\frac{K(t_0^-)}{A(t_0^-)L(t_0^-)}=k^*.
    \problemeqtag
\end{align*}
Suppose \(L(t_0^+)=mL(t_0^-)\), where \(m>1\). Capital \(K\) and technology
\(A\) cannot jump at that instant, so \(K(t_0^+)=K(t_0^-)\) and
\(A(t_0^+)=A(t_0^-)\).

\begin{enumerate}[label=(\alph*)]
    \item Immediately after the increase in workers,
    \begin{align*}
        k(t_0^+)
        &=\frac{K(t_0^+)}{A(t_0^+)L(t_0^+)}\\
        &=\frac{K(t_0^-)}{A(t_0^-)mL(t_0^-)}\\
        &=\frac{k(t_0^-)}{m}
        =\frac{k^*}{m}
        <k^*.
        \problemeqtag\label{eq:p14-impact-capital}
    \end{align*}
    Since \(y=f(k)\) and \(f'(k)>0\),
    \begin{align*}
        y(t_0^+)=f\!\left(k(t_0^+)\right)
        <f\!\left(k(t_0^-)\right)=y(t_0^-)=y^*.
        \problemeqtag
    \end{align*}
    Thus output per unit of effective labor falls immediately: the existing capital stock
    is spread across more effective labor.

    \item Because there is no population growth, the post-shock law of motion remains
    \begin{align*}
        \dot k=sf(k)-(g+\delta)k.
        \problemeqtag
    \end{align*}
    The one-time change in the level of \(L\) does not alter either curve in the Solow
    diagram. Since \eqref{eq:p14-impact-capital} shows that
    \(k(t_0^+)<k^*\),
    \begin{align*}
        sf\!\left(k(t_0^+)\right)>(g+\delta)k(t_0^+),
    \end{align*}
    so \(\dot k>0\). Capital per unit of effective labor rises toward \(k^*\), and
    \(y=f(k)\) rises with it.

    \item The balanced-growth-path condition is still
    \begin{align*}
        sf(k^*)=(g+\delta)k^*.
        \problemeqtag
    \end{align*}
    Since none of the parameters in this equation has changed, the long-run values of
    \(k^*\) and \(y^*=f(k^*)\) are unchanged. Therefore,
    \begin{align*}
        \boxed{y^*_{\mathrm{new}}=y^*_{\mathrm{old}}}
    \end{align*}
\end{enumerate}

\begin{problem}{5}
Suppose that the production function is Cobb--Douglas.
\begin{enumerate}[label=(\alph*)]
    \item Find expressions for \(k^*\), \(y^*\), and \(c^*\) as functions of the parameters
    of the model, \(s\), \(n\), \(\delta\), \(g\), and \(\alpha\).
    \item What is the golden-rule value of \(k\)?
    \item What saving rate is needed to yield the golden-rule capital stock?
\end{enumerate}
\end{problem}

\begin{solution}
Let the intensive-form production function be
\begin{align*}
    f(k)=k^\alpha,
    \qquad 0<\alpha<1.
    \problemeqtag
\end{align*}
The law of motion for capital per unit of effective labor is
\begin{align*}
    \dot k=sk^\alpha-(n+g+\delta)k.
    \problemeqtag
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item On the balanced growth path, \(\dot k=0\). Thus
    \begin{align*}
        s(k^*)^\alpha=(n+g+\delta)k^*.
        \problemeqtag
    \end{align*}
    For the positive steady state, divide by \(k^*\) and rearrange:
    \begin{align*}
        (k^*)^{1-\alpha}=\frac{s}{n+g+\delta}
    \end{align*}
    Therefore,
    \begin{align*}
        \boxed{
        k^*=\left(\frac{s}{n+g+\delta}\right)^{1/(1-\alpha)}}.
        \problemeqtag
    \end{align*}

    Since \(y=f(k)=k^\alpha\),
    \begin{align*}
        \boxed{
        y^*=\left(\frac{s}{n+g+\delta}\right)^{\alpha/(1-\alpha)}}.
        \problemeqtag
    \end{align*}
    Finally, \(c=(1-s)y\), and hence
    \begin{align*}
        \boxed{
        c^*=(1-s)
        \left(\frac{s}{n+g+\delta}\right)^{\alpha/(1-\alpha)}}.
        \problemeqtag
    \end{align*}

    \item On a balanced growth path, investment equals break-even investment.
    Consumption per unit of effective labor can therefore be written as
    \begin{align*}
        c(k)=f(k)-(n+g+\delta)k=k^\alpha-(n+g+\delta)k.
        \problemeqtag\label{eq:p15-consumption}
    \end{align*}
    The golden-rule capital stock maximizes consumption in
    \eqref{eq:p15-consumption}. The first-order condition is
    \begin{align*}
        \frac{\d c(k)}{\d k}
        =\alpha k^{\alpha-1}-(n+g+\delta)=0,
    \end{align*}
    or equivalently
    \begin{align*}
        f'(k_{\mathrm{GR}})=n+g+\delta.
        \problemeqtag\label{eq:p15-golden-rule-condition}
    \end{align*}
    Solving for \(k_{\mathrm{GR}}\) gives
    \begin{align*}
        \boxed{
        k_{\mathrm{GR}}
        =\left(\frac{\alpha}{n+g+\delta}\right)^{1/(1-\alpha)}}.
        \problemeqtag
    \end{align*}
    Moreover,
    \begin{align*}
        \frac{\d^2c(k)}{\d k^2}
        =\alpha(\alpha-1)k^{\alpha-2}<0,
    \end{align*}
    so this critical point is the unique maximum.

    \item A saving rate \(s_{\mathrm{GR}}\) supports the golden-rule capital stock if
    \begin{align*}
        s_{\mathrm{GR}}f(k_{\mathrm{GR}})
        =(n+g+\delta)k_{\mathrm{GR}}.
        \problemeqtag
    \end{align*}
    Since \(f(k)=k^\alpha\),
    \begin{align*}
        s_{\mathrm{GR}}
        =(n+g+\delta)k_{\mathrm{GR}}^{1-\alpha}
    \end{align*}
    For Cobb--Douglas production, \eqref{eq:p15-golden-rule-condition} implies
    \(k_{\mathrm{GR}}^{1-\alpha}=\alpha/(n+g+\delta)\). Hence
    \begin{align*}
        \boxed{s_{\mathrm{GR}}=\alpha}.
        \problemeqtag
    \end{align*}
    Under competitive factor pricing, \(\alpha\) is also capital's gross share of output.
\end{enumerate}
\end{solution}

\begin{note}[Steady state versus golden rule]
    The steady-state condition \(sf(k^*)=(n+g+\delta)k^*\) determines the capital
    stock generated by a given saving rate. The golden-rule condition
    \(f'(k_{\mathrm{GR}})=n+g+\delta\) instead identifies the capital stock that
    maximizes steady-state consumption.
\end{note}

\begin{note}[Cobb--Douglas shortcut]
    The result \(s_{\mathrm{GR}}=\alpha\) is specific to Cobb--Douglas production.
    Under competitive factor pricing, it says that the golden-rule saving rate equals
    capital's gross share of output.
\end{note}

\begin{problem}{6}
Consider a Solow economy that is on its balanced growth path. Assume for simplicity that there is
no technological progress. Now suppose that the rate of population growth falls.
\begin{enumerate}[label=(\alph*)]
    \item What happens to the balanced-growth-path values of capital per worker, output per worker,
    and consumption per worker? Sketch the paths of these variables as the economy moves to its new
    balanced growth path.
    \item Describe the effect of the fall in population growth on the path of output (that is,
    total output, not output per worker).
\end{enumerate}
\end{problem}

\begin{solution}
Because there is no technological progress, \(A\) is constant. We can normalize
it to one and write
\begin{align*}
    k\equiv\frac{K}{L},
    \qquad
    y\equiv\frac{Y}{L}=f(k),
    \qquad
    c\equiv\frac{C}{L}=(1-s)f(k).
    \problemeqtag
\end{align*}
Suppose the population growth rate falls permanently at \(t_0\) from
\(n_{\mathrm{old}}\) to \(n_{\mathrm{new}}<n_{\mathrm{old}}\).

\begin{enumerate}[label=(\alph*)]
    \item Capital per worker evolves according to
    \begin{align*}
        \dot k=sf(k)-(n+\delta)k.
        \problemeqtag\label{eq:p16-law-of-motion}
    \end{align*}
    Before the change, the economy is on its balanced growth path, so
    \begin{align*}
        sf\!\left(k^*_{\mathrm{old}}\right)
        =(n_{\mathrm{old}}+\delta)k^*_{\mathrm{old}}.
        \problemeqtag
    \end{align*}
    Equation \eqref{eq:p16-law-of-motion} shows that the fall in \(n\) rotates
    the break-even investment line downward from
    \((n_{\mathrm{old}}+\delta)k\) to
    \((n_{\mathrm{new}}+\delta)k\). It does not change \(sf(k)\).
    Therefore, at the old steady state,
    \begin{align*}
        \dot k(t_0^+)
        &=sf\!\left(k^*_{\mathrm{old}}\right)
          -(n_{\mathrm{new}}+\delta)k^*_{\mathrm{old}}\\
        &=(n_{\mathrm{old}}-n_{\mathrm{new}})k^*_{\mathrm{old}}>0.
        \problemeqtag
    \end{align*}
    Thus capital per worker rises monotonically until it reaches the new
    balanced-growth-path value:
    \begin{align*}
        \boxed{k^*_{\mathrm{new}}>k^*_{\mathrm{old}}}
    \end{align*}

    A change in a growth rate does not create an instantaneous level jump in
    \(K\) or \(L\). Hence
    \begin{align*}
        k(t_0^+)=k(t_0^-)=k^*_{\mathrm{old}}.
        \problemeqtag
    \end{align*}
    After \(t_0\), \(k\) rises gradually. Since \(f'(k)>0\) and \(s\) is
    unchanged, output and consumption per worker also rise gradually:
    \begin{align*}
        \boxed{
        y^*_{\mathrm{new}}>y^*_{\mathrm{old}},
        \qquad
        c^*_{\mathrm{new}}>c^*_{\mathrm{old}}}
    \end{align*}

    \item Total output satisfies \(Y=Ly\), so
    \begin{align*}
        \frac{\dot Y}{Y}
        =\frac{\dot L}{L}+\frac{\dot y}{y}
        =n+\frac{\dot y}{y}.
        \problemeqtag\label{eq:p16-output-growth}
    \end{align*}
    On the initial balanced growth path, \(\dot y=0\), so
    \eqref{eq:p16-output-growth} becomes
    \begin{align*}
        \left.\frac{\dot Y}{Y}\right|_{\mathrm{old\ BGP}}
        =n_{\mathrm{old}}.
        \problemeqtag
    \end{align*}
    On the new balanced growth path, \(\dot y=0\) again, and therefore
    \begin{align*}
        \left.\frac{\dot Y}{Y}\right|_{\mathrm{new\ BGP}}
        =n_{\mathrm{new}}.
        \problemeqtag
    \end{align*}

    There is no jump in \(Y\) at \(t_0\). Immediately after the change,
    \(K\) is still growing at its old balanced-growth-path rate
    \(n_{\mathrm{old}}\), while labor grows at \(n_{\mathrm{new}}\). Growth
    accounting therefore gives
    \begin{align*}
        \frac{\dot Y(t_0^+)}{Y(t_0^+)}
        =\alpha_K\!\left(k^*_{\mathrm{old}}\right)n_{\mathrm{old}}
        +\left[1-\alpha_K\!\left(k^*_{\mathrm{old}}\right)\right]
        n_{\mathrm{new}},
        \problemeqtag
    \end{align*}
    which lies strictly between \(n_{\mathrm{new}}\) and
    \(n_{\mathrm{old}}\). During the transition, output continues to grow,
    but its growth rate gradually approaches \(n_{\mathrm{new}}\). Hence total
    output falls increasingly below the path it would have followed if
    population growth had remained at \(n_{\mathrm{old}}\), even though output
    per worker is higher.
\end{enumerate}

\begin{problem}{7}
Find the elasticity of output per unit of effective labor on the balanced growth path, \(y^*\),
with respect to the rate of population growth, \(n\). If \(\alpha_K\left(k^*\right)=1/3\),
\(g=2\%\), and \(\delta=3\%\), by about how much does a fall in \(n\) from 2 percent to
1 percent raise \(y^*\)?
\end{problem}

\begin{solution}
The balanced-growth-path conditions are
\begin{align*}
    y^*&=f(k^*),\problemeqtag\\
    sf(k^*)&=(n+g+\delta)k^*.
    \problemeqtag\label{eq:p17-steady-state}
\end{align*}
Differentiate the steady-state condition \eqref{eq:p17-steady-state} with
respect to \(n\):
\begin{align*}
    sf'(k^*)\frac{\partial k^*}{\partial n}
    &=k^*+(n+g+\delta)\frac{\partial k^*}{\partial n}.
\end{align*}
Collecting the terms that contain \(\partial k^*/\partial n\) gives
\begin{align*}
    \left[sf'(k^*)-(n+g+\delta)\right]
    \frac{\partial k^*}{\partial n}
    =k^*,
\end{align*}
and hence
\begin{align*}
    \frac{\partial k^*}{\partial n}
    =\frac{k^*}{sf'(k^*)-(n+g+\delta)}.
\end{align*}
Use the steady-state condition to write
\begin{align*}
    s=\frac{(n+g+\delta)k^*}{f(k^*)}.
\end{align*}
Then
\begin{align*}
    sf'(k^*)
    &=(n+g+\delta)\frac{k^*f'(k^*)}{f(k^*)}\\
    &=(n+g+\delta)\alpha_K(k^*),
\end{align*}
where
\begin{align*}
    \alpha_K(k^*)\equiv\frac{k^*f'(k^*)}{f(k^*)}.
    \problemeqtag
\end{align*}
Therefore,
\begin{align*}
    \frac{\partial k^*}{\partial n}
    =-\frac{k^*}
    {(n+g+\delta)\left[1-\alpha_K(k^*)\right]}<0.
    \problemeqtag
\end{align*}

Since \(y^*=f(k^*)\), the elasticity of \(y^*\) with respect to \(n\) is
\begin{align*}
    \varepsilon_{y^*,n}
    &\equiv\frac{n}{y^*}\frac{\partial y^*}{\partial n}\\
    &=\frac{n}{f(k^*)}f'(k^*)\frac{\partial k^*}{\partial n}\\
    &=\boxed{
    -\frac{n}{n+g+\delta}
    \frac{\alpha_K(k^*)}{1-\alpha_K(k^*)}}.
    \problemeqtag\label{eq:p17-output-elasticity}
\end{align*}

For a rough calculation, evaluate \eqref{eq:p17-output-elasticity} at the midpoint
\(\bar n=1.5\%\):
\begin{align*}
    \varepsilon_{y^*,n}
    &\approx
    -\frac{0.015}{0.015+0.02+0.03}
    \frac{1/3}{1-1/3}\\
    &=-0.115\approx-0.12.
\end{align*}
The fall from \(2\%\) to \(1\%\) is a \(50\%\) fall relative to the initial
rate. The conventional textbook-style elasticity calculation therefore gives
\begin{align*}
    \frac{\Delta y^*}{y^*}
    \approx(-0.12)(-0.50)=0.06.
\end{align*}
Thus the standard rough answer is
\begin{align*}
    \boxed{y^*\text{ rises by about }6\%.}
\end{align*}

There is a small numerical subtlety because this is a large change in \(n\).
If \(\alpha_K=1/3\) is treated as constant over the interval, a
base-consistent log-change calculation gives
\begin{align*}
    \Delta\ln y^*
    &\approx-\frac{1/3}{1-1/3}
    \ln\!\left(
    \frac{0.01+0.02+0.03}{0.02+0.02+0.03}
    \right)\\
    &=\frac{1}{2}\ln\!\left(\frac{0.07}{0.06}\right)
    \approx0.077,
\end{align*}
which corresponds to an increase of about \(8\%\). The difference reflects
the crudeness of applying a local elasticity to a \(50\%\) change.
\end{solution}

\begin{note}[Local versus finite changes]
    For grading against the standard solution, the expected back-of-the-envelope
    answer is about \(6\%\). A consistent finite-change calculation is closer to
    \(8\%\) because halving \(n\) is not a small change.
\end{note}

\begin{note}[Economic intuition]
    Population growth enters through \(n+g+\delta\), so a change in \(n\) alters only
    part of the total break-even-investment rate. Plausible differences in population
    growth therefore have only modest level effects on \(y^*\).
\end{note}

\begin{problem}{8}
Suppose that investment as a fraction of output in the United States rises permanently from 0.15
to 0.18. Assume that capital's share is \(1/3\).
\begin{enumerate}[label=(\alph*)]
    \item By about how much does output eventually rise relative to what it would have been without the rise in investment?
    \item By about how much does consumption rise relative to what it would have been without the rise in investment?
    \item What is the immediate effect of the rise in investment on consumption? About how long
    does it take for consumption to return to what it would have been without the rise in investment?
\end{enumerate}
\end{problem}

\begin{solution}
Let \(s_{\mathrm{old}}=0.15\) and \(s_{\mathrm{new}}=0.18\). The proportional
increase in the investment rate is
\begin{align*}
    \frac{s_{\mathrm{new}}-s_{\mathrm{old}}}{s_{\mathrm{old}}}
    =\frac{0.03}{0.15}=0.20.
\end{align*}
Thus the saving rate rises by \(20\%\), not by \(3\%\); the latter number is
the change in percentage points.

\begin{enumerate}[label=(\alph*)]
    \item From equation (1.28) in the text, the balanced-growth-path elasticity
    of output with respect to the saving rate is
    \begin{align*}
        \varepsilon_{y^*,s}
        \equiv\frac{s}{y^*}\frac{\partial y^*}{\partial s}
        =\frac{\alpha_K(k^*)}{1-\alpha_K(k^*)}.
        \problemeqtag\label{eq:p18-output-elasticity}
    \end{align*}
    With \(\alpha_K=1/3\), \eqref{eq:p18-output-elasticity} becomes
    \begin{align*}
        \varepsilon_{y^*,s}
        =\frac{1/3}{1-1/3}=\frac{1}{2}
    \end{align*}
    Hence the elasticity approximation gives
    \begin{align*}
        \frac{\Delta y^*}{y^*}
        \approx\frac{1}{2}(0.20)=0.10
    \end{align*}
    Therefore,
    \begin{align*}
        \boxed{\text{Long-run output is about }10\%\text{ higher.}}
    \end{align*}
    If a Cobb--Douglas production function with capital share \(1/3\) is used
    to calculate the finite change exactly, then
    \begin{align*}
        \frac{y^*_{\mathrm{new}}}{y^*_{\mathrm{old}}}
        =\left(\frac{s_{\mathrm{new}}}{s_{\mathrm{old}}}\right)^{(1/3)/(2/3)}
        =\sqrt{1.2}
        \approx1.095,
    \end{align*}
    so the exact increase is \(9.5\%\), which confirms the approximation.

    \item Balanced-growth-path consumption per unit of effective labor is
    \begin{align*}
        c^*=(1-s)y^*.
        \problemeqtag\label{eq:p18-steady-consumption}
    \end{align*}
    Therefore, using \eqref{eq:p18-steady-consumption},
    \begin{align*}
        \frac{c^*_{\mathrm{new}}}{c^*_{\mathrm{old}}}
        &=\frac{1-s_{\mathrm{new}}}{1-s_{\mathrm{old}}}
          \frac{y^*_{\mathrm{new}}}{y^*_{\mathrm{old}}}\\
        &=\frac{0.82}{0.85}\sqrt{1.2}\\
        &\approx1.057
    \end{align*}
    Thus
    \begin{align*}
        \boxed{\text{Long-run consumption is about }6\%\text{ higher.}}
    \end{align*}
    Consumption rises less than output because the economy devotes a larger
    fraction of its higher output to investment.

    \item At the instant the saving rate rises, \(K\), \(A\), \(L\), and hence
    \(y\), cannot jump. Consumption does jump because the consumed fraction of
    output falls immediately:
    \begin{align*}
        \frac{c(t_0^+)}{c(t_0^-)}
        &=\frac{(1-s_{\mathrm{new}})y(t_0)}
                {(1-s_{\mathrm{old}})y(t_0)}\\
        &=\frac{0.82}{0.85}
        \approx0.965
    \end{align*}
    Hence
    \begin{align*}
        \boxed{\text{Consumption falls immediately by about }3.5\%.}
    \end{align*}

    To estimate the recovery time, use the text's benchmark
    \(n+g+\delta\approx0.06\). The local convergence rate is
    \begin{align*}
        \lambda
        &=\left[1-\alpha_K(k^*)\right](n+g+\delta)\\
        &=\left(1-\frac13\right)(0.06)=0.04.
        \problemeqtag
    \end{align*}
    Normalize consumption on the old path to one, and define
    \begin{align*}
        x_0&\equiv\frac{c(t_0^+)}{c^*_{\mathrm{old}}}
        =\frac{0.82}{0.85}\approx0.965,\\
        x^*_{\mathrm{new}}
        &\equiv\frac{c^*_{\mathrm{new}}}{c^*_{\mathrm{old}}}
        \approx1.057
    \end{align*}
    The linearized transition is
    \begin{align*}
        x\left(t\right)-x^*_{\mathrm{new}}
        \approx e^{-\lambda(t-t_0)}
        \left(x_0-x^*_{\mathrm{new}}\right).
        \problemeqtag\label{eq:p18-consumption-transition}
\end{align*}
    Consumption has returned to its old path when \(x\left(t\right)=1\). Substituting
    this condition into \eqref{eq:p18-consumption-transition} gives
    \begin{align*}
        e^{-\lambda(t-t_0)}
        &=\frac{x^*_{\mathrm{new}}-1}
        {x^*_{\mathrm{new}}-x_0}\\
        &\approx\frac{1.057-1}{1.057-0.965}
        \approx0.62
    \end{align*}
    Taking logs gives
    \begin{align*}
        t-t_0
        &\approx-\frac{\ln(0.62)}{0.04}\\
        &\approx12\text{ years}
    \end{align*}
    Therefore,
    \begin{align*}
        \boxed{\text{Consumption returns to its old path after roughly }12\text{ years.}}
    \end{align*}
\end{enumerate}

\begin{problem}{9}[\textbf{Factor payments in the Solow model.}]
Assume that both labor and capital are paid their marginal products. Let \(w\) denote
\(\partial F\left(K,AL\right)/\partial L\) and \(r\) denote
\(\left[\partial F\left(K,AL\right)/\partial K\right]-\delta\).
\begin{enumerate}[label=(\alph*)]
    \item Show that the marginal product of labor, \(w\), is \(A\left[f\left(k\right)-k\,f'\left(k\right)\right]\).
    \item Show that if both capital and labor are paid their marginal products, constant returns
    to scale imply that the total amount paid to the factors of production equals total net output.
    That is, show that under constant returns, \(wL+rK=F\left(K,AL\right)-\delta K\).
    \item The return to capital (\(r\)) is roughly constant over time, as are the shares of output
    going to capital and to labor. Does a Solow economy on a balanced growth path exhibit these
    properties? What are the growth rates of \(w\) and \(r\) on a balanced growth path?
    \item Suppose the economy begins with a level of \(k\) less than \(k^*\). As \(k\) moves
    toward \(k^*\), is \(w\) growing at a rate greater than, less than, or equal to its growth
    rate on the balanced growth path? What about \(r\)?
\end{enumerate}
\end{problem}

\begin{solution}
Write the constant-returns production function in intensive form:
\begin{align*}
    Y=F(K,AL)=ALf(k),
    \qquad
    k\equiv\frac{K}{AL}.
    \problemeqtag
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item The wage is the marginal product of labor, holding \(K\) and \(A\)
    fixed. First,
    \begin{align*}
        \frac{\partial k}{\partial L}
        =\frac{\partial}{\partial L}\left(\frac{K}{AL}\right)
        =-\frac{K}{AL^2}
        =-\frac{k}{L}
    \end{align*}
    Apply the product and chain rules to \(Y=ALf(k)\):
    \begin{align*}
        w
        \equiv\frac{\partial Y}{\partial L}
        &=A f(k)+ALf'(k)\frac{\partial k}{\partial L}\\
        &=A f(k)+ALf'(k)\left(-\frac{k}{L}\right)\\
        &=\boxed{A\left[f(k)-kf'(k)\right]}.
        \problemeqtag\label{eq:p19-wage}
    \end{align*}

    \item First derive the marginal product of capital. Since
    \begin{align*}
        \frac{\partial k}{\partial K}=\frac{1}{AL},
    \end{align*}
    we have
    \begin{align*}
        \frac{\partial Y}{\partial K}
        =ALf'(k)\frac{1}{AL}=f'(k).
        \problemeqtag
    \end{align*}
    Therefore the net return to capital is
    \begin{align*}
        r=f'(k)-\delta.
        \problemeqtag
    \end{align*}
    Total labor income is
    \begin{align*}
        wL
        &=AL\left[f(k)-kf'(k)\right]\\
        &=ALf(k)-ALk f'(k)\\
        &=Y-Kf'(k),
    \end{align*}
    where \(ALk=K\). Total net capital income is
    \begin{align*}
        rK=Kf'(k)-\delta K
    \end{align*}
    Adding the two payments gives
    \begin{align*}
        wL+rK
        &=Y-Kf'(k)+Kf'(k)-\delta K\\
        &=\boxed{F(K,AL)-\delta K}.
        \problemeqtag\label{eq:p19-factor-exhaustion}
    \end{align*}
    Equation \eqref{eq:p19-factor-exhaustion} is Euler's theorem for a
    constant-returns production function, with
    depreciation deducted from capital's payment.

    \item On a balanced growth path, \(k=k^*\) is constant. Hence
    \begin{align*}
        r=f'(k^*)-\delta
        \problemeqtag
    \end{align*}
    is constant, so
    \begin{align*}
        \boxed{\frac{\dot r}{r}=0}.
        \problemeqtag
    \end{align*}
    Evaluating the wage formula \eqref{eq:p19-wage} at \(k^*\) gives
    \begin{align*}
        w=A\left[f(k^*)-k^*f'(k^*)\right].
        \problemeqtag
    \end{align*}
    The expression in brackets is constant on the balanced growth path, while
    \(A\) grows at rate \(g\). Therefore,
    \begin{align*}
        \boxed{\frac{\dot w}{w}=g}.
        \problemeqtag
    \end{align*}

    The factor shares are constant as well. For example, \(r\) is constant,
    and both \(K\) and \(Y\) grow at \(n+g\), so
    \begin{align*}
        \frac{\d}{\d t}\ln\!\left(\frac{rK}{Y}\right)
        =0+(n+g)-(n+g)=0.
        \problemeqtag
    \end{align*}
    Likewise, \(w\) grows at \(g\), \(L\) grows at \(n\), and \(Y\) grows at
    \(n+g\), so \(wL/Y\) is constant. The shares in net output are also
    constant because \(Y-\delta K\), \(wL\), and \(rK\) all grow at
    \(n+g\) on the balanced growth path.

    \item Let
    \begin{align*}
        h(k)\equiv f(k)-kf'(k),
        \problemeqtag
    \end{align*}
    so that \(w=Ah(k)\). Its derivative is
    \begin{align*}
        h'(k)
        =f'(k)-\left[f'(k)+kf''(k)\right]
        =-kf''(k)
    \end{align*}
    Thus
    \begin{align*}
        \frac{\dot w}{w}
        &=\frac{\dot A}{A}+\frac{h'(k)\dot k}{h(k)}\\
        &=g-\frac{kf''(k)\dot k}{f(k)-kf'(k)}.
        \problemeqtag
    \end{align*}
    When \(k<k^*\), stability implies \(\dot k>0\). With diminishing marginal
    returns, \(f''(k)<0\), and labor's gross marginal product satisfies
    \(f(k)-kf'(k)>0\). Therefore,
    \begin{align*}
        \boxed{\frac{\dot w}{w}>g}
    \end{align*}
    The wage grows faster than on the balanced growth path because workers
    benefit from both technological progress and capital deepening.

    For the net return to capital,
    \begin{align*}
        r=f'(k)-\delta,
        \qquad
        \dot r=f''(k)\dot k<0.
        \problemeqtag
    \end{align*}
    If \(r>0\), its proportional growth rate is
    \begin{align*}
        \frac{\dot r}{r}
        =\frac{f''(k)\dot k}{f'(k)-\delta}<0.
        \problemeqtag
    \end{align*}
    Hence
    \begin{align*}
        \boxed{r\text{ falls during the transition, so its growth rate is below }0.}
    \end{align*}
    It approaches the constant balanced-growth-path return from above.
\end{enumerate}
\end{solution}

\begin{note}[Net versus gross return]
    The problem defines \(r=f'(k)-\delta\), the \emph{net} return. Therefore
    \(\dot r/r=f''(k)\dot k/[f'(k)-\delta]\); using \(f'(k)\) in the denominator
    would instead be the growth rate of the gross marginal product of capital.
\end{note}

\begin{problem}{10}
\problemsource{\parencite{piketty2014capital,piketty-zucman2014capital,rognlie2015deciphering}}
This problem asks you to use a Solow-style model to investigate some ideas that have been discussed
in the context of Thomas Piketty's recent work. Consider an economy described by the assumptions of the Solow model, except that
factors are paid their marginal products (as in the previous problem), and all labor income is
consumed and all other income is saved. Thus,
\(C\left(t\right)=L\left(t\right)\left[\partial Y\left(t\right)/\partial L\left(t\right)\right]\).
\begin{enumerate}[label=(\alph*)]
    \item Show that the properties of the production function and our assumptions about the behavior
    of \(L\) and \(A\) imply that the capital-output ratio, \(K/Y\), is rising if and only if
    the growth rate of \(K\) is greater than \(n+g\)--that is, if and only if \(k\) is rising.
    \item Assume that the initial conditions are such that \(\partial Y/\partial K\) at \(t=0\)
    is strictly greater than \(n+g+\delta\). Describe the qualitative behavior of the capital-output
    ratio over time. (For example, does it grow or fall without bound? Gradually approach some
    constant level from above or below? Something else?) Explain your reasoning.
    \item Many popular summaries of Piketty's work describe his thesis as: Since the return to
    capital exceeds the growth rate of the economy, the capital-output ratio tends to grow without
    bound. By the assumptions in part (b), this economy starts in a situation where the return to
    capital exceeds the economy's growth rate. If you found in (b) that \(K/Y\) grows without
    bound, explain intuitively whether the driving force of this unbounded growth is that the return
    to capital exceeds the economy's growth rate. Alternatively, if you found in (b) that \(K/Y\)
    does not grow without bound, explain intuitively what is wrong with the statement that the return
    to capital exceeding the economy's growth rate tends to cause \(K/Y\) to grow without bound.
    \item Suppose \(F\left(\,\cdot\,\right)\) is Cobb--Douglas and that the initial situation is as in
    part (b). Describe the qualitative behavior over time of the share of net capital income (that is,
    \(K\left(t\right)\left[\partial Y\left(t\right)/\partial K\left(t\right)-\delta\right]\))
    in net output (that is, \(Y\left(t\right)-\delta K\left(t\right)\)). Explain your reasoning.
    Is the common statement that an excess of the return to capital over the economy's growth rate
    causes capital's share to rise over time correct in this case?
\end{enumerate}
\end{problem}

\begin{solution}
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{4pt}
Let \(F_K\equiv\partial F(K,AL)/\partial K\), and define capital's gross
output elasticity by
\begin{align*}
    \alpha_K\left(t\right)\equiv\frac{F_K\left(t\right)K\left(t\right)}{Y\left(t\right)}.
    \problemeqtag
\end{align*}
Because the production function has constant returns to scale, labor income
plus gross capital income exhausts output. Since all labor income is consumed,
\begin{align*}
    Y-C=F_KK.
    \problemeqtag
\end{align*}
The capital accumulation equation is therefore
\begin{align*}
    \dot K
    &=Y-C-\delta K\\
    &=\left(F_K-\delta\right)K.
    \problemeqtag\label{eq:p110-capital-accumulation}
\end{align*}
Thus the growth rate of capital equals the net return to capital:
\begin{align*}
    \frac{\dot K}{K}=F_K-\delta.
    \problemeqtag
\end{align*}

\begin{enumerate}[label=(\alph*)]
    \item Differentiate \(Y=F(K,AL)\) with respect to time:
    \begin{align*}
        \dot Y=F_K\dot K+F_{AL}\,\dot{(AL)}
    \end{align*}
    Divide by \(Y\), multiply and divide the first term by \(K\), and multiply
    and divide the second term by \(AL\):
    \begin{align*}
        \frac{\dot Y}{Y}
        &=\frac{F_KK}{Y}\frac{\dot K}{K}
        +\frac{F_{AL}(AL)}{Y}\frac{\dot{(AL)}}{AL}
    \end{align*}
    Constant returns imply
    \begin{align*}
        \frac{F_KK}{Y}+\frac{F_{AL}(AL)}{Y}=1,
        \problemeqtag
    \end{align*}
    and \(AL\) grows at \(n+g\). Hence
    \begin{align*}
        \frac{\dot Y}{Y}
        =\alpha_K\frac{\dot K}{K}
        +(1-\alpha_K)(n+g).
        \problemeqtag
    \end{align*}
    The growth rate of the capital-output ratio is therefore
    \begin{align*}
        \frac{\d}{\d t}\ln\!\left(\frac{K}{Y}\right)
        &=\frac{\dot K}{K}-\frac{\dot Y}{Y}\\
        &=(1-\alpha_K)
        \left[\frac{\dot K}{K}-(n+g)\right].
        \problemeqtag
    \end{align*}
    With \(0<\alpha_K<1\), this expression is positive if and only if
    \begin{align*}
        \frac{\dot K}{K}>n+g
    \end{align*}
    Finally, since \(k=K/(AL)\),
    \begin{align*}
        \frac{\dot k}{k}=\frac{\dot K}{K}-(n+g).
        \problemeqtag
    \end{align*}
    Thus
    \begin{align*}
        \boxed{
        \frac{K}{Y}\text{ rises}
        \quad\Longleftrightarrow\quad
        \frac{\dot K}{K}>n+g
        \quad\Longleftrightarrow\quad
        \dot k>0.}
    \end{align*}

    \item In intensive form, \(F_K=f'(k)\). Using the accumulation equation
    \eqref{eq:p110-capital-accumulation},
    \begin{align*}
        \frac{\dot k}{k}
        &=\frac{\dot K}{K}-(n+g)\\
        &=f'(k)-(n+g+\delta).
        \problemeqtag
    \end{align*}
    The assumption in the question is
    \begin{align*}
        f'(k(0))>n+g+\delta,
    \end{align*}
    so \(\dot k(0)>0\). Under the usual Solow assumptions,
    \(f''(k)<0\), so the marginal product \(f'(k)\) falls as \(k\) rises.
    Let \(\bar k\) denote the unique value satisfying
    \begin{align*}
        f'(\bar k)=n+g+\delta.
        \problemeqtag
    \end{align*}
    Then
    \begin{align*}
        k<\bar k&\Longrightarrow\dot k>0,\\
        k>\bar k&\Longrightarrow\dot k<0
    \end{align*}
    Since the initial condition implies \(k(0)<\bar k\), capital per unit of
    effective labor rises monotonically toward \(\bar k\).

    Moreover,
    \begin{align*}
        \frac{K}{Y}=\frac{k}{f(k)},
        \problemeqtag\label{eq:p110-capital-output-level}
    \end{align*}
    and
    \begin{align*}
        \frac{\d\ln[k/f(k)]}{\d\ln k}
        =1-\frac{kf'(k)}{f(k)}
        =1-\alpha_K(k)>0.
        \problemeqtag\label{eq:p110-capital-output-elasticity}
    \end{align*}
    Therefore \(K/Y\) rises whenever \(k\) rises and converges to the finite
    value \(\bar k/f(\bar k)\). Hence
    \begin{align*}
        \boxed{\frac{K}{Y}\text{ approaches a finite constant from below.}}
    \end{align*}

    \item At the start of the transition,
    \begin{align*}
        r\equiv F_K-\delta>n+g.
        \problemeqtag
    \end{align*}
    Since \(\dot K/K=r\), capital initially grows faster than effective labor,
    and \(K/Y\) rises. But the return \(r\) is not fixed: as \(k\) rises,
    diminishing marginal returns reduce \(F_K=f'(k)\), and hence reduce \(r\).
    The process stops when
    \begin{align*}
        r=n+g,
        \problemeqtag
    \end{align*}
    equivalently when \(f'(k)=n+g+\delta\). At that point \(k\) and \(K/Y\)
    are constant.

    Thus the statement that \(r\) exceeding the economy's growth rate causes
    \(K/Y\) to grow without bound incorrectly treats the return to capital as
    exogenous. In this model,
    \begin{align*}
        \boxed{\text{capital accumulation lowers }r\text{ until the excess return disappears.}}
    \end{align*}

    \item With Cobb--Douglas production,
    \begin{align*}
        Y=K^\alpha(AL)^{1-\alpha},
        \qquad 0<\alpha<1,
        \problemeqtag
    \end{align*}
    so
    \begin{align*}
        F_K=\alpha\frac{Y}{K}.
        \problemeqtag
    \end{align*}
    Net capital income is
    \begin{align*}
        K(F_K-\delta)
        =\alpha Y-\delta K,
        \problemeqtag
    \end{align*}
    while net output is \(Y-\delta K\). Let
    \begin{align*}
        \beta\equiv\frac{K}{Y}.
        \problemeqtag
    \end{align*}
    The share of net capital income in net output is
    \begin{align*}
        \theta_K^{\mathrm{net}}
        &\equiv\frac{\alpha Y-\delta K}{Y-\delta K}\\
        &=\frac{\alpha-\delta\beta}{1-\delta\beta}.
        \problemeqtag
    \end{align*}
    Differentiate with respect to \(\beta\):
    \begin{align*}
        \frac{\d\theta_K^{\mathrm{net}}}{\d\beta}
        &=\frac{-\delta(1-\delta\beta)
        -(\alpha-\delta\beta)(-\delta)}{(1-\delta\beta)^2}\\
        &=\frac{\delta(\alpha-1)}{(1-\delta\beta)^2}<0.
        \problemeqtag
    \end{align*}
    Equations \eqref{eq:p110-capital-output-level} and
    \eqref{eq:p110-capital-output-elasticity} show that \(\beta=K/Y\) rises
    monotonically. Therefore,
    \begin{align*}
        \boxed{\theta_K^{\mathrm{net}}\text{ falls monotonically toward a constant.}}
    \end{align*}
    Thus, in the Cobb--Douglas case, an initial excess of the return to capital
    over the economy's growth rate is associated with a \emph{fall}, not a rise,
    in capital's share of net output.
\end{enumerate}

\begin{problem}{11}
Consider Problem 1.10. Suppose there is a marginal increase in \(K\).
\begin{enumerate}[label=(\alph*)]
    \item Derive an expression (in terms of \(K/Y\), \(\delta\), the marginal product of capital
    \(F_K\), and the elasticity of substitution between capital and effective labor in the gross
    production function \(F\left(\,\cdot\,\right)\)) that determines whether a marginal increase
    in \(K\) increases, reduces, or has no effect on the share of net capital income in net output.
    \item Suppose the capital-output ratio is 3, \(\delta=3\%\), and the rate of return on capital
    (\(F_K-\delta\)) = 5\%. How large must the elasticity of substitution in the gross production
    function be for the share of net capital income in net output to rise when \(K\) rises?
\end{enumerate}
\end{problem}

\begin{solution}
Let
\begin{align*}
    \theta_K^{\mathrm{net}}
    &\equiv \frac{K(F_K-\delta)}{Y-\delta K}
    =\frac{F_K-\delta}{Y/K-\delta}
    \problemeqtag\label{eq:p111-net-share}
\end{align*}
denote capital's share of net output. We maintain the usual assumptions that
\(F_K>\delta\), \(Y/K>\delta\), and \(F_{KK}<0\).

\begin{enumerate}[label=(\alph*)]
    \item Along a constant-returns-to-scale production function, the elasticity
    of substitution between capital and effective labor can be written as
    \begin{align*}
        \sigma
        &=-\frac{\partial\ln(K/Y)}{\partial\ln F_K}
        =\frac{\partial\ln(Y/K)}{\partial\ln F_K}.
        \problemeqtag\label{eq:p111-sigma}
    \end{align*}
    Hence
    \begin{align*}
        \frac{\partial(Y/K)}{\partial F_K}
        =\sigma\frac{Y/K}{F_K}.
        \problemeqtag\label{eq:p111-output-capital-derivative}
    \end{align*}

    Differentiate \eqref{eq:p111-net-share} with respect to \(F_K\):
    \begin{align*}
        \frac{\partial\theta_K^{\mathrm{net}}}{\partial F_K}
        &=\frac{1}{Y/K-\delta}
        -\frac{F_K-\delta}{(Y/K-\delta)^2}
        \frac{\partial(Y/K)}{\partial F_K}\\
        &=\frac{(Y/K-\delta)
        -\sigma(F_K-\delta)(Y/K)/F_K}
        {(Y/K-\delta)^2}.
        \problemeqtag\label{eq:p111-share-derivative}
    \end{align*}
    Since \(F_K\) falls when \(K\) rises, the sign of the response to \(K\)
    is the opposite of the sign in \eqref{eq:p111-share-derivative}. Therefore,
    \begin{align*}
        \boxed{
        \begin{array}{rcl}
        \theta_K^{\mathrm{net}}\text{ rises with }K
        &\Longleftrightarrow&
        \displaystyle \sigma>
        \frac{F_K[1-\delta(K/Y)]}{F_K-\delta},\\[7pt]
        \theta_K^{\mathrm{net}}\text{ is locally unchanged}
        &\Longleftrightarrow&
        \displaystyle \sigma=
        \frac{F_K[1-\delta(K/Y)]}{F_K-\delta},\\[7pt]
        \theta_K^{\mathrm{net}}\text{ falls with }K
        &\Longleftrightarrow&
        \displaystyle \sigma<
        \frac{F_K[1-\delta(K/Y)]}{F_K-\delta}.
        \end{array}}
        \problemeqtag\label{eq:p111-threshold}
    \end{align*}

    \item The net return is \(F_K-\delta=0.05\), so
    \(F_K=0.08\). Substituting \(K/Y=3\) and \(\delta=0.03\) into
    \eqref{eq:p111-threshold} gives
    \begin{align*}
        \sigma
        &>\frac{0.08[1-(0.03)(3)]}{0.08-0.03}
        =\frac{0.0728}{0.05}
        =1.456
    \end{align*}
    Thus
    \begin{align*}
        \boxed{\sigma>1.456}
    \end{align*}
    is required for a marginal increase in \(K\) to raise capital's share of
    net output.
\end{enumerate}
\end{solution}

\begin{note}[Sign check]
    The derivative in \eqref{eq:p111-share-derivative} is taken with respect
    to \(F_K\), not \(K\). Diminishing marginal returns imply
    \(\partial F_K/\partial K<0\), so the final inequality reverses.
\end{note}

\begin{note}[Gross versus net]
    \(F_K\) is the gross marginal product, whereas \(F_K-\delta\) is the net
    return. Likewise, the relevant denominator is net output \(Y-\delta K\),
    not gross output \(Y\).
\end{note}

\begin{problem}{12}
Consider the same setup as at the start of Problem 1.10: the economy is described by the assumptions
of the Solow model, except that factors are paid their marginal products and all labor income is
consumed and all other income is saved. Show that the economy converges to a balanced growth path,
and that the balanced-growth-path level of \(k\) equals the golden-rule level of \(k\). What is
the intuition for this result?
\end{problem}

\begin{solution}
Constant returns to scale and marginal-product factor pricing imply that gross
capital income is \(F_KK\). Since all labor income is consumed and all other
income is saved, capital accumulation satisfies
\begin{align*}
    \dot K=(F_K-\delta)K.
    \problemeqtag\label{eq:p112-capital-law}
\end{align*}
Let \(k=K/(AL)\). Because \(AL\) grows at rate \(n+g\) and
\(F_K=f'(k)\), equation \eqref{eq:p112-capital-law} yields
\begin{align*}
    \dot k
    =\bigl[f'(k)-(n+g+\delta)\bigr]k.
    \problemeqtag\label{eq:p112-intensive-law}
\end{align*}
Apart from the zero-capital steady state, the balanced-growth-path value
\(k^*>0\) is determined by
\begin{align*}
    f'(k^*)=n+g+\delta.
    \problemeqtag\label{eq:p112-bgp}
\end{align*}
Since \(f''(k)<0\),
\begin{align*}
    k<k^*&\Longrightarrow f'(k)>n+g+\delta
    \Longrightarrow \dot k>0,\\
    k>k^*&\Longrightarrow f'(k)<n+g+\delta
    \Longrightarrow \dot k<0.
\end{align*}
Thus every path with positive initial capital converges to \(k^*\). Once
\(k\) is constant, both \(K\) and \(Y\) grow at rate \(n+g\), so the limiting
path is a balanced growth path.

Consumption per unit of effective labor at any constant \(k\) is
\begin{align*}
    c(k)=f(k)-(n+g+\delta)k.
\end{align*}
The golden-rule capital stock maximizes \(c(k)\), and its first-order condition is
\begin{align*}
    f'(k_{\mathrm{GR}})=n+g+\delta.
    \problemeqtag\label{eq:p112-golden-rule}
\end{align*}
Equations \eqref{eq:p112-bgp} and \eqref{eq:p112-golden-rule} are identical;
hence
\begin{align*}
    \boxed{k^*=k_{\mathrm{GR}}.}
\end{align*}

The intuition follows directly from saving behavior. Saving per unit of
effective labor is capital's gross income, \(f'(k)k\), while break-even
investment is \((n+g+\delta)k\). Capital rises when the former exceeds the
latter and falls when it is smaller. The adjustment ends precisely where the
marginal product of capital equals the resource cost of maintaining one more
unit of \(k\), which is the golden-rule condition.
\end{solution}

\begin{note}[Interpretation]
    The saving rate is endogenous in this model: households save all capital
    income and consume all labor income. It should not be treated as the fixed
    \(s\) of the basic Solow model.
\end{note}

\begin{problem}{13}
Go through steps analogous to those in equations (1.29)--(1.32) to find how quickly \(y\) converges
to \(y^*\) in the vicinity of the balanced growth path.
\par \remarkaccent{Hint:} Since \(y=f\left(k\right)\), we can write \(k=g\left(y\right)\), where \(g\left(\,\cdot\,\right)=f^{-1}\left(\,\cdot\,\right)\).
\end{problem}

\begin{solution}
Let \(k=g(y)=f^{-1}(y)\). Differentiating \(y=f(k)\) and using the Solow
law of motion gives
\begin{align*}
    \dot y
    &=f'(k)\dot k\\
    &=f'(k)\bigl[sf(k)-(n+g+\delta)k\bigr]
    \equiv H(y).
    \problemeqtag\label{eq:p113-output-law}
\end{align*}
At \(y^*=f(k^*)\), \(H(y^*)=0\). A first-order approximation around
\(y^*\) is therefore
\begin{align*}
    \dot y\left(t\right)
    \simeq H'(y^*)[y\left(t\right)-y^*]
    \equiv-\lambda_y[y\left(t\right)-y^*].
    \problemeqtag\label{eq:p113-linearization}
\end{align*}
The solution to the linearized equation is
\begin{align*}
    y\left(t\right)-y^*
    \simeq e^{-\lambda_y t}[y(0)-y^*].
    \problemeqtag\label{eq:p113-path}
\end{align*}

It remains to calculate \(H'(y^*)\). By the chain rule,
\begin{align*}
    H'(y^*)
    &=\left.
    \frac{\partial\dot y}{\partial k}
    \frac{\partial k}{\partial y}
    \right|_{k=k^*}\\
    &=\left.
    \left\{
    f''(k)[sf(k)-(n+g+\delta)k]
    +f'(k)[sf'(k)-(n+g+\delta)]
    \right\}\frac{1}{f'(k)}
    \right|_{k=k^*}\\
    &=sf'(k^*)-(n+g+\delta),
    \problemeqtag\label{eq:p113-derivative}
\end{align*}
where the steady-state condition
\(sf(k^*)=(n+g+\delta)k^*\) eliminates the first term.
\Needspace{7\baselineskip}
Consequently,
\begin{align*}
    \lambda_y
    &=(n+g+\delta)-sf'(k^*)\\
    &=(n+g+\delta)
    \left[1-\frac{k^*f'(k^*)}{f(k^*)}\right]\\
    &=\bigl[1-\alpha_K(k^*)\bigr](n+g+\delta).
    \problemeqtag\label{eq:p113-convergence-rate}
\end{align*}
Thus
\begin{align*}
    \boxed{\lambda_y=\lambda_k
    =\bigl[1-\alpha_K(k^*)\bigr](n+g+\delta).}
\end{align*}
Output per unit of effective labor therefore converges locally at exactly the
same rate as capital per unit of effective labor.
\end{solution}

\begin{note}[Economic intuition]
    Near the balanced growth path, \(y-y^*\simeq
    f'(k^*)(k-k^*)\). This local proportionality is why the two gaps shrink at
    the same exponential rate even though \(y\) and \(k\) have different units.
\end{note}

\begin{problem}{14}[\textbf{Embodied technological progress.}]
\problemsource{\parencite{solow1960investment,sato1966adjustment}}
One view of technological progress is that the
productivity of capital goods built at \(t\) depends on the state of technology at \(t\) and is
unaffected by subsequent technological progress. This is known as \emph{embodied technological
progress} (technological progress must be ``embodied'' in new capital before it can raise output).
This problem asks you to investigate its effects.
\begin{enumerate}[label=(\alph*)]
    \item As a preliminary, let us modify the basic Solow model to make technological progress
    capital-augmenting rather than labor-augmenting. So that a balanced growth path exists, assume
    that the production function is Cobb--Douglas:
    \begin{align*}
    Y\left(t\right)
    &=\left[A\left(t\right)K\left(t\right)\right]^{\alpha}L\left(t\right)^{1-\alpha}
    \end{align*}
    Assume that \(A\) grows at rate \(\mu\): \(\dot{A}\left(t\right)=\mu A\left(t\right)\).
    Show that the economy converges to a balanced growth path, and find the growth rates of \(Y\)
    and \(K\) on the balanced growth path.
    \par \remarkaccent{Hint:} Show that we can write
    \(Y/\left(A^{\phi}L\right)\) as a function of \(K/\left(A^{\phi}L\right)\), where
    \(\phi=\alpha/\left(1-\alpha\right)\). Then analyze the dynamics of
    \(K/\left(A^{\phi}L\right)\).
    \item Now consider embodied technological progress. Specifically, let the production function be
    \(Y\left(t\right)=J\left(t\right)^{\alpha}L\left(t\right)^{1-\alpha}\), where \(J\left(t\right)\)
    is the effective capital stock. The dynamics of \(J\left(t\right)\) are given by
    \(\dot{J}\left(t\right)=s\,A\left(t\right)Y\left(t\right)-\delta J\left(t\right)\).
    The presence of the \(A\left(t\right)\) term in this expression means that the productivity
    of investment at \(t\) depends on the technology at \(t\). Show that the economy converges
    to a balanced growth path. What are the growth rates of \(Y\) and \(J\) on the balanced
    growth path?
    \par \remarkaccent{Hint:} Let \(\bar{J}\left(t\right)\equiv J\left(t\right)/A\left(t\right)\). Then use the same approach as in (a), focusing on \(\bar{J}/\left(A^{\phi}L\right)\) instead of \(K/\left(A^{\phi}\,L\right)\).
    \item What is the elasticity of output on the balanced growth path with respect to \(s\)?
    \item In the vicinity of the balanced growth path, how rapidly does the economy converge to the balanced growth path?
    \item Compare your results for (c) and (d) with the corresponding results in the text for
    the basic Solow model.
\end{enumerate}
\end{problem}

\begin{solution}
Let
\[
    \phi\equiv\frac{\alpha}{1-\alpha}.
\]

\begin{enumerate}[label=(\alph*)]
    \item Define capital and output relative to \(A^\phi L\):
    \[
        \kappa\left(t\right)\equiv\frac{K\left(t\right)}{A\left(t\right)^\phi L\left(t\right)},
        \qquad
        y\left(t\right)\equiv\frac{Y\left(t\right)}{A\left(t\right)^\phi L\left(t\right)}.
    \]
    Since \(1+\phi=1/(1-\alpha)\), dividing the production function by
    \(A^\phi L\) gives
    \begin{align*}
        y\left(t\right)=\kappa\left(t\right)^\alpha.
        \problemeqtag\label{eq:p114-capital-augmenting-production}
    \end{align*}
    The normalizing factor \(A^\phi L\) grows at rate \(\phi\mu+n\).
    Using \(\dot K=sY-\delta K\), the law of motion for \(\kappa\) is
    \begin{align*}
        \dot\kappa
        =s\kappa^\alpha-(n+\phi\mu+\delta)\kappa.
        \problemeqtag\label{eq:p114-kappa-law}
    \end{align*}
    The unique positive steady state is
    \begin{align*}
        \kappa^*
        =\left(\frac{s}{n+\phi\mu+\delta}\right)^{1/(1-\alpha)}.
        \problemeqtag\label{eq:p114-kappa-star}
    \end{align*}
    For \(0<\kappa<\kappa^*\), actual investment exceeds break-even
    investment and \(\dot\kappa>0\); for \(\kappa>\kappa^*\), the reverse
    holds. Thus \(\kappa\to\kappa^*\) and, by
    \eqref{eq:p114-capital-augmenting-production}, \(y\to y^*\).

    Since \(K=A^\phi L\kappa\) and \(Y=A^\phi Ly\), the balanced-growth-path
    growth rates are
    \[
        \boxed{g_K^{\mathrm{BGP}}=g_Y^{\mathrm{BGP}}=n+\phi\mu.}
    \]

    \item Under embodied technological progress, define
    \(\bar J\equiv J/A\). Its law of motion is
    \begin{align*}
        \dot{\bar J}
        &=\frac{\dot J}{A}-\mu\bar J
        =sY-(\delta+\mu)\bar J.
        \problemeqtag\label{eq:p114-jbar-law}
    \end{align*}
    The production function becomes
    \(Y=(A\bar J)^\alpha L^{1-\alpha}\), which has the same form as in
    part (a). Define
    \[
        j\left(t\right)\equiv\frac{\bar J\left(t\right)}{A\left(t\right)^\phi L\left(t\right)}
        =\frac{J\left(t\right)}{A\left(t\right)^{1+\phi}L\left(t\right)}.
    \]
    Then
    \begin{align*}
        y\left(t\right)=j\left(t\right)^\alpha,
        \qquad
        y\left(t\right)\equiv\frac{Y\left(t\right)}{A\left(t\right)^\phi L\left(t\right)}.
        \problemeqtag\label{eq:p114-embodied-production}
    \end{align*}
    Combining \eqref{eq:p114-jbar-law} with the growth rate of
    \(A^\phi L\) yields
    \begin{align*}
        \dot j
        =sj^\alpha-\bigl[n+\delta+(1+\phi)\mu\bigr]j.
        \problemeqtag\label{eq:p114-j-law}
    \end{align*}
    Thus the unique positive steady state is
    \begin{align*}
        j^*
        =\left(
        \frac{s}{n+\delta+(1+\phi)\mu}
        \right)^{1/(1-\alpha)},
        \problemeqtag\label{eq:p114-j-star}
    \end{align*}
    and the same sign argument as in part (a) shows that \(j\to j^*\).

    On the balanced growth path, \(y\) and \(j\) are constant. Hence
    \[
        \boxed{g_Y^{\mathrm{BGP}}=n+\phi\mu,}
        \qquad
        \boxed{g_J^{\mathrm{BGP}}=n+(1+\phi)\mu.}
    \]
    The growth rate of \(J\) is higher because \(J=A^{1+\phi}Lj\), whereas
    output is \(Y=A^\phi Ly\).

    \item From \eqref{eq:p114-embodied-production} and
    \eqref{eq:p114-j-star},
    \[
        y^*=\left(
        \frac{s}{n+\delta+(1+\phi)\mu}
        \right)^{\alpha/(1-\alpha)}.
    \]
    Since \(Y=A^\phi Ly\), the elasticity of balanced-growth-path output
    with respect to \(s\) is
    \begin{align*}
        \boxed{
        \frac{\partial\ln Y^{\mathrm{BGP}}}{\partial\ln s}
        =\frac{\partial\ln y^*}{\partial\ln s}
        =\frac{\alpha}{1-\alpha}.}
        \problemeqtag\label{eq:p114-saving-elasticity}
    \end{align*}

    \item Let
    \(B\equiv n+\delta+(1+\phi)\mu\). Linearizing
    \eqref{eq:p114-j-law} around \(j^*\) gives
    \[
        \dot j\simeq
        \bigl[s\alpha(j^*)^{\alpha-1}-B\bigr](j-j^*).
    \]
    The steady-state condition implies
    \(s(j^*)^{\alpha-1}=B\). Therefore,
    \begin{align*}
        \boxed{
        \lambda_{\mathrm{emb}}
        =(1-\alpha)\bigl[n+\delta+(1+\phi)\mu\bigr].}
        \problemeqtag\label{eq:p114-convergence}
    \end{align*}
    Locally,
    \[
        j\left(t\right)-j^*\simeq e^{-\lambda_{\mathrm{emb}}t}[j(0)-j^*].
    \]
    Because \(y=j^\alpha\) is smooth and strictly increasing, \(y\) converges
    at the same rate.

    \item For Cobb--Douglas production, the basic Solow model gives
    \[
        \frac{\partial\ln y^*}{\partial\ln s}
        =\frac{\alpha}{1-\alpha},
        \qquad
        \lambda_{\mathrm{basic}}=(1-\alpha)(n+\mu+\delta).
    \]
    Thus embodiment leaves the saving elasticity unchanged. It increases the
    convergence rate, since
    \[
        \lambda_{\mathrm{emb}}-\lambda_{\mathrm{basic}}
        =\alpha\mu>0.
    \]
\end{enumerate}

\begin{problem}{15}
Consider a Solow economy on its balanced growth path. Suppose the growth-accounting techniques
described in Section 1.7 are applied to this economy.
\begin{enumerate}[label=(\alph*)]
    \item What fraction of growth in output per worker does growth accounting attribute to growth
    in capital per worker? What fraction does it attribute to technological progress?
    \item How can you reconcile your results in (a) with the fact that the Solow model implies that
    the growth rate of output per worker on the balanced growth path is determined solely by the rate
    of technological progress?
\end{enumerate}
\end{problem}

\begin{solution}
Let \(\alpha_K\left(t\right)\equiv F_K\left(t\right)K\left(t\right)/Y\left(t\right)\) denote capital's output
elasticity. Growth accounting for output per worker gives
\begin{align*}
    \frac{\dot Y}{Y}-\frac{\dot L}{L}
    =\alpha_K\left(t\right)
    \left(\frac{\dot K}{K}-\frac{\dot L}{L}\right)
    +R\left(t\right),
    \problemeqtag\label{eq:p115-growth-accounting}
\end{align*}
where \(R\left(t\right)\) is the Solow residual.

\begin{enumerate}[label=(\alph*)]
    \item On the Solow balanced growth path,
    \begin{align*}
        \frac{\dot Y}{Y}-\frac{\dot L}{L}
        =\frac{\dot K}{K}-\frac{\dot L}{L}
        =g.
        \problemeqtag\label{eq:p115-bgp-growth}
    \end{align*}
    Substituting \eqref{eq:p115-bgp-growth} into
    \eqref{eq:p115-growth-accounting} gives
    \begin{align*}
        g=\alpha_K(k^*)g+R,
        \qquad
        R=[1-\alpha_K(k^*)]g.
        \problemeqtag\label{eq:p115-decomposition}
    \end{align*}
    Thus growth accounting attributes
    \begin{align*}
        \boxed{\alpha_K(k^*)\text{ of growth to capital deepening}}
    \end{align*}
    and
    \begin{align*}
        \boxed{1-\alpha_K(k^*)\text{ of growth to technological progress}.}
    \end{align*}
    With the common calibration \(\alpha_K=1/3\), the shares are one-third
    and two-thirds, respectively.

    \item There is no contradiction because growth accounting records
    \emph{proximate} contributions, whereas the Solow model identifies the
    \emph{underlying} source of sustained growth. Growth in \(A\) raises output
    per worker directly. For a fixed saving rate, it also induces capital per
    worker to grow at rate \(g\), so part of technology's total effect operates
    through capital deepening. Growth accounting assigns this induced channel
    to capital rather than to the technological change that caused it.

    Hence the accounting decomposition is correct, but it should not be read as
    a causal decomposition of the ultimate sources of balanced-growth-path
    growth.
\end{enumerate}
\end{solution}

\begin{note}[Accounting versus causality]
    The term \(\alpha_K g\) answers how much current capital deepening
    contributes mechanically to output growth. It does not answer why capital
    deepening occurs.
\end{note}

\begin{note}[Solow residual]
    Labor-augmenting technology grows at rate \(g\), but the residual in
    \eqref{eq:p115-decomposition} is \((1-\alpha_K)g\). The remaining
    \(\alpha_K g\) appears in the capital-deepening term.
\end{note}

\begin{problem}{16}
\begin{enumerate}[label=(\alph*)]
    \item In the model of convergence and measurement error in equations (1.39) and (1.40), suppose
    the true value of \(b\) is \(-1\). Does a regression of
    \(\ln\left(Y/N\right)_{1979}-\ln\left(Y/N\right)_{1870}\) on a constant and
    \(\ln\left(Y/N\right)_{1870}\) yield a biased estimate of \(b\)? Explain.
    \item Suppose there is measurement error in measured 1979 income per capita but not in 1870
    income per capita. Does a regression of
    \(\ln\left(Y/N\right)_{1979}-\ln\left(Y/N\right)_{1870}\) on a constant and
    \(\ln\left(Y/N\right)_{1870}\) yield a biased estimate of \(b\)? Explain.
\end{enumerate}
\end{problem}

\begin{solution}
Write true log income per capita in 1870 as \(x_i^*\), measured log income
as \(x_i\), and true 1979 log income as \(z_i^*\).

\begin{enumerate}[label=(\alph*)]
    \item Equations (1.39) and (1.40) are
    \begin{align*}
        z_i^*-x_i^*&=a+bx_i^*+\varepsilon_i,
        \problemeqtag\label{eq:p116-true-model}\\
        x_i&=x_i^*+u_i,
        \problemeqtag\label{eq:p116-initial-measurement}
    \end{align*}
    where \(u_i\) is uncorrelated with \(x_i^*\) and
    \(\varepsilon_i\). The regression uses measured growth
    \(z_i^*-x_i\). Substituting \(x_i^*=x_i-u_i\) into
    \eqref{eq:p116-true-model} gives
    \begin{align*}
        z_i^*-x_i
        =a+bx_i+\varepsilon_i-(1+b)u_i.
        \problemeqtag\label{eq:p116-observed-regression}
    \end{align*}
    In general, \(x_i=x_i^*+u_i\) is correlated with the composite error.
    Indeed,
    \begin{align*}
        \operatorname{Cov}
        \left(x_i,\varepsilon_i-(1+b)u_i\right)
        =-(1+b)\operatorname{Var}(u_i).
        \problemeqtag\label{eq:p116-error-covariance}
    \end{align*}
    When the true \(b=-1\), however, the measurement-error term disappears
    from \eqref{eq:p116-observed-regression}; the regression error is simply
    \(\varepsilon_i\), which is uncorrelated with \(x_i\). Therefore,
    \begin{align*}
        \boxed{b=-1\Longrightarrow\text{the OLS slope is unbiased}.}
    \end{align*}

    \item Now suppose 1870 income is measured without error, so \(x_i=x_i^*\),
    but measured 1979 income is \(z_i=z_i^*+v_i\). Measured growth satisfies
    \begin{align*}
        z_i-x_i
        =a+bx_i+\varepsilon_i+v_i.
        \problemeqtag\label{eq:p116-late-measurement}
    \end{align*}
    Under classical measurement error,
    \(\operatorname{E}[v_i\mid x_i]=0\). Thus the new composite error
    \(\varepsilon_i+v_i\) remains uncorrelated with the regressor, and
    \begin{align*}
        \boxed{\text{measurement error only in 1979 income does not bias }\hat b.}
    \end{align*}
    It raises the variance of the regression error and therefore reduces
    precision.
\end{enumerate}
\end{solution}

\begin{note}[Why initial-income error is special]
    Measured 1870 income enters both sides with opposite signs. A positive
    error makes a country look richer initially and slower growing, mechanically
    favoring convergence unless \(b=-1\).
\end{note}

\begin{note}[Classical-error qualification]
    Part (b) requires mean-zero 1979 error uncorrelated with initial income and
    the structural disturbance. Systematic error can still bias OLS.
\end{note}

\begin{problem}{17}
Derive equation (1.50).
\par \remarkaccent{Hint:} Follow steps analogous to those in equations [1.47] and [1.48].
\end{problem}

\begin{solution}
\setlength{\abovedisplayskip}{6pt}
\setlength{\belowdisplayskip}{6pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{4pt}
The hypothetical economy removes resource and land limitations by assuming
that resources, land, and labor all grow at rate \(n\):
\[
    g_R=g_T=g_L=n,
    \qquad g_A=g.
\]
The capital accumulation equation gives \(g_K=s(Y/K)-\delta\). A constant
balanced-growth-path value of \(g_K\) therefore requires \(Y/K\) to be
constant, and hence
\begin{align*}
    g_K^{\mathrm{BGP}}=g_Y^{\mathrm{BGP}}.
    \problemeqtag\label{eq:p117-equal-growth}
\end{align*}

Log-differentiating the production function gives
\begin{align*}
    g_Y
    =\alpha g_K+\beta g_R+\gamma g_T
    +(1-\alpha-\beta-\gamma)(g_A+g_L).
    \problemeqtag\label{eq:p117-growth-accounting}
\end{align*}
Substitute \(g_R=g_T=g_L=n\) and \(g_A=g\):
\begin{align*}
    g_Y
    &=\alpha g_K+(\beta+\gamma)n
    +(1-\alpha-\beta-\gamma)(n+g)\\
    &=\alpha g_K+(1-\alpha)n
    +(1-\alpha-\beta-\gamma)g.
    \problemeqtag\label{eq:p117-no-limits-growth}
\end{align*}
Imposing \eqref{eq:p117-equal-growth} yields
\begin{align*}
    (1-\alpha)\widetilde g_Y^{\mathrm{BGP}}
    &=(1-\alpha)n+(1-\alpha-\beta-\gamma)g,\\
    \widetilde g_Y^{\mathrm{BGP}}
    &=n+\frac{1-\alpha-\beta-\gamma}{1-\alpha}g.
    \problemeqtag\label{eq:p117-total-growth}
\end{align*}
The tilde distinguishes the hypothetical economy from the economy with fixed
land and declining resource use.
Output per worker grows at total-output growth minus population growth.
Therefore,
\begin{align*}
    \boxed{
    \widetilde g_{Y/L}^{\mathrm{BGP}}
    =\widetilde g_Y^{\mathrm{BGP}}-n
    =\frac{1-\alpha-\beta-\gamma}{1-\alpha}g.}
    \problemeqtag
\end{align*}
This is equation (1.50).
\end{solution}
