\newcommand{\courseproblemjump}[3]{%
    \@ifundefined{CourseProblemJumpList@#1}{%
        \expandafter\gdef\csname CourseProblemJumpList@#1\endcsname{%
            \courseproblemjumpitem{#2}{#3}%
        }%
    }{%
        \expandafter\g@addto@macro\csname CourseProblemJumpList@#1\endcsname{%
            \problemjumpsep\courseproblemjumpitem{#2}{#3}%
        }%
    }%
}

\newcommand{\recordcourseproblemjump}{%
    \begingroup
        \edef\course@unit{\CurrentCourseUnit}%

        \protected@write\@auxout{}{%
            \string\courseproblemjump{\course@unit}{\course@number}{\course@target}%
        }%
    \endgroup
}

% ============================================================
% 十、题目、答案、评注与注释环境
% problem 和 answer 使用带编号的悬挂标签；solution 是 answer 的
% 别名；remark 跟随当前题号；note 不带题号且正文与普通正文对齐。
% 可选标题仅在实际提供时显示，并在标题后添加句点。
% ============================================================
\NewDocumentEnvironment{problem}{m O{}}
    {%
        \Needspace{5\baselineskip}%
        \renewcommand{\CurrentProblemLocalNumber}{#1}%
        \renewcommand{\CurrentProblemNumber}{\CurrentCourseUnit.#1}%
        \refstepcounter{problem}%
        \par\vspace{14pt}%
        \hypertarget{problem-\theHproblem}{}%
        \recordcourseproblemjump
        \blocklabel{ProblemAccent}{\ProblemLabel}{\CurrentProblemNumber}%
        \ifblank{#2}{}{%
            \hspace{0.45em}{\normalsize\itshape\color{Muted}#2}%
        }%
        \par\vspace{2pt}%
        \begingroup
        \ignorespaces
    }
    {%
        \par\endgroup
        \ignorespacesafterend
    }

\NewDocumentEnvironment{answer}{O{}}
    {%
        \Needspace{5\baselineskip}%
        \par\vspace{4pt}%
        \blocklabel{AnswerAccent}{\AnswerLabel}{\CurrentProblemNumber}%
        \ifblank{#1}{}{%
            \hspace{0.45em}{\normalsize\itshape\color{Muted}#1}%
        }%
        \par\vspace{2pt}%
        \begingroup
        \ignorespaces
    }
    {%
        \par\endgroup
        \ignorespacesafterend
    }

\NewDocumentEnvironment{solution}{O{}}

    {\end{answer}}

\NewDocumentEnvironment{remark}{O{}}
    {%

            \ifblank{#1}{}{~#1.}}%
        \hspace{0.45em}\small\color{Muted}\ignorespaces
    }
    {%
        \par\endgroup
        \ignorespacesafterend
    }
